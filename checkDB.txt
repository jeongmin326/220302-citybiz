**사용자**
-- citybizdb.users definition
CREATE TABLE `users` (
  `user_id` bigint(20) NOT NULL AUTO_INCREMENT COMMENT '사용자 고유 ID (PK)',
  `email` varchar(255) NOT NULL COMMENT '로그인 이메일 (아이디 역할)',
  `password_hash` varchar(255) NOT NULL COMMENT '비밀번호 해시 값',
  `name` varchar(50) NOT NULL COMMENT '사용자 이름',
  `phone` varchar(20) DEFAULT NULL COMMENT '연락처 (선택 입력)',
  `role` varchar(20) NOT NULL DEFAULT 'USER' COMMENT '사용자 권한 (USER / ADMIN)',
  `business_stage` varchar(30) DEFAULT NULL COMMENT '사업 단계 (예비창업자, 스타트업 등)',
  `status` varchar(20) NOT NULL DEFAULT 'ACTIVE' COMMENT '계정 상태 (ACTIVE / INACTIVE / SUSPENDED)',
  `created_at` datetime NOT NULL DEFAULT current_timestamp() COMMENT '계정 생성일시',
  `updated_at` datetime NOT NULL DEFAULT current_timestamp() ON UPDATE current_timestamp() COMMENT '계정 수정일시',
  `company_name` varchar(100) DEFAULT NULL COMMENT '기업명 또는 팀명',
  `biz_no` varchar(30) DEFAULT NULL COMMENT '사업자 등록번호',
  `industry` varchar(50) DEFAULT NULL COMMENT '업종 또는 관심 분야',
  PRIMARY KEY (`user_id`),
  UNIQUE KEY `uk_users_email` (`email`),
  UNIQUE KEY `uq_users_phone` (`phone`)

**공간**
-- citybizdb.spaces definition
CREATE TABLE `spaces` (
  `space_id` bigint(20) NOT NULL AUTO_INCREMENT COMMENT '공간 고유 ID',
  `host_id` bigint(20) DEFAULT NULL COMMENT '공간 등록자(호스트) user_id',
  `name` varchar(100) NOT NULL COMMENT '공간 이름',
  `region` varchar(50) NOT NULL COMMENT '광역 지역(예: 서울, 경기)',
  `district` varchar(50) NOT NULL COMMENT '세부 지역(예: 강남구, 성수동)',
  `address` varchar(255) NOT NULL COMMENT '상세 주소',
  `space_type` varchar(30) NOT NULL COMMENT '공간 유형(shop, warehouse, studio, meeting, consulting, office)',
  `price_per_hour` int(11) NOT NULL COMMENT '시간당 가격',
  `capacity` int(11) NOT NULL COMMENT '최대 수용 인원',
  `main_image_url` varchar(500) DEFAULT NULL COMMENT '대표 이미지 경로',
  `description` text DEFAULT NULL COMMENT '공간 설명',
  `available_yn` char(1) NOT NULL DEFAULT 'Y' COMMENT '사용 가능 여부(Y/N)',
  `created_at` datetime NOT NULL DEFAULT current_timestamp() COMMENT '등록일시',
  `updated_at` datetime NOT NULL DEFAULT current_timestamp() ON UPDATE current_timestamp() COMMENT '수정일시',
  PRIMARY KEY (`space_id`),
  KEY `idx_spaces_host_id` (`host_id`),
  CONSTRAINT `fk_spaces_host` FOREIGN KEY (`host_id`) REFERENCES `users` (`user_id`) ON DELETE SET NULL
) ENGINE=InnoDB AUTO_INCREMENT=91 DEFAULT CHARSET=utf8mb4 COMMENT='공간 기본 정보 테이블';
