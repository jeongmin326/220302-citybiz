use publicservice;

CREATE TABLE users (
    user_id BIGINT AUTO_INCREMENT PRIMARY KEY 
        COMMENT '사용자 고유 ID (기본키, 자동 증가)',
    email VARCHAR(100) NOT NULL UNIQUE 
        COMMENT '로그인에 사용하는 이메일 (아이디 역할, 중복 불가)',
    password_hash VARCHAR(255) NOT NULL 
        COMMENT '비밀번호 저장용 컬럼 (현재는 평문, 추후 암호화 필요)',
    name VARCHAR(50) NOT NULL 
        COMMENT '사용자 이름',
    phone VARCHAR(20) NULL 
        COMMENT '사용자 연락처 (선택 입력)',
    role VARCHAR(20) NOT NULL DEFAULT 'USER' 
        COMMENT '사용자 권한 (USER: 일반 사용자, ADMIN: 관리자)',
    business_stage VARCHAR(30) NULL 
        COMMENT '사업 단계 (예비창업자, 스타트업, 소상공인 등 구분)',
    status VARCHAR(20) NOT NULL DEFAULT 'ACTIVE' 
        COMMENT '계정 상태 (ACTIVE: 사용 가능, INACTIVE: 비활성, SUSPENDED: 정지)',
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP 
        COMMENT '회원 가입 시간',
    updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP 
        ON UPDATE CURRENT_TIMESTAMP 
        COMMENT '회원 정보 수정 시간'
) COMMENT='사용자 계정 정보를 저장하는 테이블';

use publicservice;

CREATE TABLE spaces (
    space_id BIGINT AUTO_INCREMENT PRIMARY KEY
        COMMENT '공간 고유 ID',
    name VARCHAR(100) NOT NULL
        COMMENT '공간 이름',
    region VARCHAR(50) NOT NULL
        COMMENT '광역 지역(예: 서울, 경기)',
    district VARCHAR(50) NOT NULL
        COMMENT '세부 지역(예: 강남구, 성수동)',
    address VARCHAR(255) NOT NULL
        COMMENT '상세 주소',
    space_type VARCHAR(30) NOT NULL
        COMMENT '공간 유형(shop, warehouse, studio, meeting, consulting, office)',
    price_per_hour INT NOT NULL
        COMMENT '시간당 가격',
    capacity INT NOT NULL
        COMMENT '최대 수용 인원',
    main_image_url VARCHAR(500) NULL
        COMMENT '대표 이미지 경로',
    description TEXT NULL
        COMMENT '공간 설명',
    available_yn CHAR(1) NOT NULL DEFAULT 'Y'
        COMMENT '사용 가능 여부(Y/N)',
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
        COMMENT '등록일시',
    updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP
        COMMENT '수정일시'
) COMMENT='공간 기본 정보 테이블';


CREATE TABLE space_images (
    image_id BIGINT AUTO_INCREMENT PRIMARY KEY
        COMMENT '이미지 고유 ID',
    space_id BIGINT NOT NULL
        COMMENT '공간 ID',
    image_url VARCHAR(500) NOT NULL
        COMMENT '이미지 경로',
    is_main CHAR(1) NOT NULL DEFAULT 'N'
        COMMENT '대표 이미지 여부(Y/N)',
    sort_order INT NOT NULL DEFAULT 0
        COMMENT '이미지 정렬 순서',
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
        COMMENT '등록일시',
    CONSTRAINT fk_space_images_space
        FOREIGN KEY (space_id) REFERENCES spaces(space_id)
        ON DELETE CASCADE
) COMMENT='공간 이미지 테이블';

CREATE INDEX idx_spaces_region ON spaces(region);
CREATE INDEX idx_spaces_district ON spaces(district);
CREATE INDEX idx_spaces_space_type ON spaces(space_type);
CREATE INDEX idx_spaces_price_per_hour ON spaces(price_per_hour);
CREATE INDEX idx_spaces_capacity ON spaces(capacity);
CREATE INDEX idx_spaces_available_yn ON spaces(available_yn);

CREATE INDEX idx_space_images_space_id ON space_images(space_id);
CREATE INDEX idx_space_images_sort_order ON space_images(sort_order);
